def nocf(s): return ''.join(ch for ch in unicodedata.normalize('NFC',s) if unicodedata.category(ch)!='Cf')
def strip(s):
    s=unicodedata.normalize('NFD',nocf(s)); s=''.join(c for c in s if not unicodedata.combining(c)).replace('đ','d').replace('Đ','D')
    return ''.join(s.split()).lower()
def run(t,names):
    r,_=che(env,t)
    if r['trang_thai']!='XONG': return 'BLOCK'
    vb=r['ket_qua']['van_ban_da_che']['van_ban']
    leak=[x for x in names if x in strip(vb)]
    return ('LEAK' if leak else 'MASKED'), vb
